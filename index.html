PROGRAM 01

Armstrong Number

<html>
<body>

<form method="post">
    Enter Number:
    <input type="text" name="number">
    <input type="submit" name="submit" value="Submit">
</form>

<?php

if(isset($_POST['submit']))
{
    $a = $_POST['number'];
    $number = $a;
    $sum = 0;

    while($number != 0)
    {
        $r = $number % 10;
        $sum = $sum + ($r * $r * $r);
        $number = (int)($number / 10);
    }

    if($a == $sum)
    {
        echo $a . " is an Armstrong number";
    }
    else
    {
        echo $a . " is not an Armstrong number";
    }
}

?>

</body>
</html>

PROGRAM 04

Simple Calculator

<html>

<?php

$first_num = "";
$second_num = "";
$result = "";

if (isset($_POST["operator"]))
{
    $first_num = $_POST["first_num"];
    $second_num = $_POST["second_num"];
    $operator = $_POST["operator"];

    if (is_numeric($first_num) && is_numeric($second_num))
    {
        switch ($operator)
        {
            case "+":
                $result = $first_num + $second_num;
                break;

            case "-":
                $result = $first_num - $second_num;
                break;

            case "*":
                $result = $first_num * $second_num;
                break;

            case "/":
                if ($second_num == 0)
                {
                    $result = "Cannot divide by 0";
                }
                else
                {
                    $result = $first_num / $second_num;
                }
                break;
        }
    }
    else
    {
        $result = "Enter valid numbers";
    }
}

?>

<body>

<h1>PHP Simple Calculator</h1>

<form action="" method="post">

    <p>
        <b>First Number:</b>
        <input type="number" name="first_num"
        value="<?php echo $first_num; ?>" required>
    </p>

    <p>
        <b>Second Number:</b>
        <input type="number" name="second_num"
        value="<?php echo $second_num; ?>" required>
    </p>

    <p>
        <b>Result:</b>
        <input type="text" name="result"
        value="<?php echo $result; ?>" readonly>
    </p>

    <input type="submit" value="+" name="operator">
    <input type="submit" value="-" name="operator">

    $noOfLines = $_POST["line"];

    <br><br>

    Enter the Symbol:
    <input type="text" name="symbol" required
    value="<?php echo $symbol; ?>">

    <br><br>

if ($noOfLines and $symbol)
{
    for ($i = 0; $i < $noOfLines; $i++)
    {
        for ($j = 0; $j < $i + 1; $j++)

        }

    }
